cmake_minimum_required(VERSION 3.18)
project(DroneRoutePlanning LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(navigation src/navigation.cpp)
target_include_directories(navigation PUBLIC include)
add_executable(drone_route src/main.cpp)
target_link_libraries(drone_route PRIVATE navigation)
enable_testing()
add_executable(test_navigation tests/test_navigation.cpp)
target_link_libraries(test_navigation PRIVATE navigation)
add_test(NAME navigation_test COMMAND test_navigation)
